# Вызов конструктора класса-дедушки

# Определим класс-дедушку
class Grandparent:
    def __init__(self):
        print("Конструктор дедушки")

# Класс-родитель, наследует Grandparent
class Parent(Grandparent):
    def __init__(self):
        print("Конструктор родителя")
        # Вызов конструктора через super()
        super().__init__()  

# Класс-наследник, наследует Parent
class Child(Parent):
    def __init__(self):
        print("Конструктор наследника")
        # Конструктор класса-дедушки напрямую
        Grandparent.__init__(self)

# Создаем экземпляр класса Child
child_instance = Child()
